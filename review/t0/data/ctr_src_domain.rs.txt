/// 沿 chain 解析出来的最终设置。会话第一个任务把它锁进 Session.config_snapshot（CT09）。
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub struct EffectiveSettings {
    /// 从 org 到叶子
    pub chain: Vec<ScopeRef>,
    pub enabled: bool,
    pub respond_automatically: bool,
    pub network: SandboxNetwork,
    /// 空 = 主模型
    #[serde(default)]
    pub model: String,
    pub external_chat_mode: ExternalChatMode,
    pub search_scope: SearchScope,
    #[serde(default)]
    pub spend_limit_cny_month: Option<f64>,
    #[serde(default)]
    pub instructions: String,
    /// 合并后的访问包
    #[serde(default)]
    pub bundle: AccessBundle,
    /// = compute_snapshot_hash()
    #[serde(default)]
    pub snapshot_hash: String,
}

impl EffectiveSettings {
    /// sha256(canonical_json(除 snapshot_hash 外的全部字段)) 的 hex。与 snapshot_hash 当前值无关。
    pub fn compute_snapshot_hash(&self) -> String {
        let mut body = match serde_json::to_value(self) {
            Ok(Value::Object(m)) => m,
            _ => Map::new(),
        };
        body.remove("snapshot_hash");
        hex::encode(Sha256::digest(canonical_json(&body).as_bytes()))
    }
}

// ---------------------------------------------------------------- routines

str_enum! {
    ScheduleKind { Once => "once", Interval => "interval", Cron => "cron" }
}

fn default_tz_name() -> String {
    "Asia/Shanghai".to_string()
}

fn default_tz_offset_min() -> i32 {
    480
}

#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Schedule {
    pub kind: ScheduleKind,
    /// once
    #[serde(default)]
    pub at: Option<DateTime<Utc>>,
    /// interval
    #[serde(default)]
    pub every_min: Option<u32>,
    /// cron（5 段）
    #[serde(default)]
    pub cron: Option<String>,
    #[serde(default = "default_tz_name")]
    pub tz_name: String,
    #[serde(default = "default_tz_offset_min")]
    pub tz_offset_min: i32,
}

/// 例程怎么被触发。JSON 形如 {"kind": "schedule", "schedule": {...}}。
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
pub enum RoutineTrigger {
    Schedule {
        schedule: Schedule,
    },
    /// 盯某群：出现关键词就触发
    ChannelWatch {
        chat_id: String,
        keyword: String,
    },
    /// 外部事件：ExternalEvent.topic 相等就触发
    External {
        topic: String,
    },
}

/// 例程的产出发到哪。JSON 形如 {"kind": "chat", "chat_id": "..."}。
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
#[serde(tag = "kind", rename_all = "snake_case")]
pub enum RoutineOutput {
    Chat {
        chat_id: String,
        #[serde(default)]
        thread_id: Option<String>,
    },
    Dm {
        user_id: String,
    },
    /// 托管页；doc_id 有值 = 每次更新同一页
    Page {
        #[serde(default)]
        doc_id: Option<String>,
    },
}

// ---------------------------------------------------------------- services

/// 五个领域存储的句柄包：control / worker / gateway 各持一份；没接上的就是 None。
#[derive(Clone, Default)]
pub struct Services {
    pub scope: Option<Arc<dyn ScopeStore>>,
    pub memory: Option<Arc<dyn MemoryStore>>,
    pub routines: Option<Arc<dyn RoutineStore>>,
    pub usage: Option<Arc<dyn UsageLedger>>,
    pub audit: Option<Arc<dyn AuditLog>>,
}

impl std::fmt::Debug for Services {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.debug_struct("Services")
            .field("scope", &self.scope.is_some())
            .field("memory", &self.memory.is_some())
            .field("routines", &self.routines.is_some())
            .field("usage", &self.usage.is_some())
            .field("audit", &self.audit.is_some())
            .finish()
    }
}
