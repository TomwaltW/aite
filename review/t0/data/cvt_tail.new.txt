#[test]
fn network_levels_and_egress_policy() {
    for (s, want) in [
        ("", SandboxNetwork::None),
        ("none", SandboxNetwork::None),
        ("trusted", SandboxNetwork::Trusted),
        ("custom", SandboxNetwork::Custom),
        ("full", SandboxNetwork::Full),
    ] {
        let spec = SandboxSpec::try_from(pb::SandboxSpec {
            image: "img:1".into(),
            network: s.into(),
            ..Default::default()
        })
        .unwrap();
        assert_eq!(spec.network, want, "{s:?}");
    }
    let mut spec = SandboxSpec::new("img:1");
    spec.network = SandboxNetwork::Trusted;
    assert_eq!(
        SandboxSpec::try_from(pb::SandboxSpec::from(spec.clone())).unwrap(),
        spec
    );
    let pol = EgressPolicy {
        level: SandboxNetwork::Custom,
        allow_hosts: vec!["*.example.test".into(), "git.example.test:8443".into()],
        credentials: vec![CredentialBinding {
            host: "git.example.test:8443".into(),
            header: "Authorization".into(),
            scheme: "Bearer".into(),
            secret_ref: "gitlab_bot".into(),
        }],
        audit_tag: "t1".into(),
    };
    let p = pb::EgressPolicy::from(pol.clone());
    assert_eq!(p.level, "custom");
    assert_eq!(EgressPolicy::try_from(p).unwrap(), pol);
    assert_eq!(
        EgressPolicy::try_from(pb::EgressPolicy::default()).unwrap(),
        EgressPolicy::default(),
        "空 level = none"
    );
    assert!(
        EgressPolicy::try_from(pb::EgressPolicy {
            level: "bridge".into(),
            ..Default::default()
        })
        .is_err()
    );
}
