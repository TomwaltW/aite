#[derive(Debug, thiserror::Error)]
pub enum ModelError {
    /// 缺 base_url / model / 密钥环境变量。消息里只出现变量名，绝不出现取值。
    #[error("模型配置不完整：{0}")]
    Config(String),
    #[error("模型服务错误：{0}")]
    Upstream(String),
    #[error("模型响应无法解析：{0}")]
    BadResponse(String),
    /// 429：retry_after_ms 取自 Retry-After（没给就是 None，调用方按退避策略）
    #[error("模型服务限流：{message}")]
    RateLimited {
        retry_after_ms: Option<u64>,
        message: String,
    },
    /// 401 / 403：密钥错或没权限，重试没用
    #[error("模型服务鉴权失败：{0}")]
    Auth(String),
    /// 400：请求本身被拒（参数、上下文超长……），重试没用
    #[error("模型服务拒绝请求：{0}")]
    BadRequest(String),
}
