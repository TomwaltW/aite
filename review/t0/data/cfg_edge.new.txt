/// core ↔ edge 的进程边界（proto/aite/v1/edge.proto）。socket 路径相对仓库根。
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct EdgeConfig {
    /// edge 监听：PlatformService / SandboxService / EdgeStatusService
    pub edge_socket: String,
    /// core 监听：IngressService
    pub core_socket: String,
    /// edge 调 HandleEvent 的 deadline（§3.3 "on_event 1s 内返回"）
    pub handle_event_deadline_ms: u64,
    /// 两边 gRPC 的最大消息体（文件走 bytes）
    pub max_message_mb: u32,
    /// edge 的 HTTP 入口（外部 webhook）；空 = 不开
    pub http_listen: String,
    pub webhook_secret_env: String,
}

impl Default for EdgeConfig {
    fn default() -> Self {
        Self {
            edge_socket: "data/run/aite-edge.sock".into(),
            core_socket: "data/run/aite-core.sock".into(),
            handle_event_deadline_ms: 1000,
            max_message_mb: 64,
            http_listen: String::new(),
            webhook_secret_env: "AITE_WEBHOOK_SECRET".into(),
        }
    }
}

/// 访问控制（CT28 / NEW06 / NEW08）。
#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
#[serde(default, deny_unknown_fields)]
pub struct AccessConfig {
    pub dm_enabled: bool,
    /// 外部群：restrict = 回固定提示、不执行；离开 restrict 要 compliance 的两个确认位都为 true（D25）

    /// 非空 = 只在这些群里工作

    pub block_chats: Vec<String>,

    pub search_scope: SearchScope,
    /// true：群管理员 !configure 之前不接活
    pub require_setup: bool,
}

impl Default for AccessConfig {
    fn default() -> Self {
        Self {
            dm_enabled: true,
            external_chat_mode: ExternalChatMode::Restrict,
            allow_chats: Vec::new(),
            block_chats: Vec::new(),
            member_allowlist: Vec::new(),
            search_scope: SearchScope::JoinedChats,
            require_setup: false,
        }
    }
}
