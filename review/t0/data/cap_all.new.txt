#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct PlatformCapabilities {
    /// "feishu" | "dingtalk" | "wecom" | "fake"
    pub platform: String,
    /// 有原生线程/话题
    pub supports_thread: bool,
    /// 能读群历史
    pub supports_history: bool,
    /// 能收到非 @ 消息
    pub supports_passive_listen: bool,
    /// 卡片能原地更新
    pub supports_card_edit: bool,
    /// 卡片可更新时长（飞书 14 天 = 1209600）
    pub card_edit_window_sec: u32,
    /// 群里能收文件
    pub inbound_file_in_group: bool,
    pub proactive_requires_prior_message: bool,
    /// 出站限速（消息数/分钟），adapter 自己令牌桶
    pub outbound_rate_per_min: u32,
    // —— p1.0 新增（proto 字段 10–23）。数值 0 = 未声明，调用方按保守值处理 ——
    /// 能收到表情回应事件
    #[serde(default)]
    pub supports_reactions_in: bool,
    /// 能收到撤回事件
    #[serde(default)]
    pub supports_recall_event: bool,
    /// 能收到编辑事件（飞书没有：编辑靠拉历史比对）
    #[serde(default)]
    pub supports_edit_event: bool,
    #[serde(default)]
    pub supports_pins: bool,
    #[serde(default)]
    pub supports_message_search: bool,
    /// 能写平台云文档（托管页）
    #[serde(default)]
    pub supports_docs: bool,
    #[serde(default)]
    pub supports_dm: bool,
    /// 引用带被引消息的 id
    #[serde(default)]
    pub supports_quote_id: bool,
    /// 没有话题：每条回复都要带可见的 #A 编号
    #[serde(default)]
    pub requires_visible_anchor: bool,
    /// 流式卡片最长持续秒数
    #[serde(default)]
    pub stream_max_sec: u32,
    /// 卡片回调必须在这么多毫秒内应答
    #[serde(default)]
    pub card_action_deadline_ms: u32,
    #[serde(default)]
    pub max_card_bytes: u32,
    #[serde(default)]
    pub max_text_chars: u32,
    #[serde(default)]
    pub max_upload_bytes: u64,
}
