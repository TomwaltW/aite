#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Turn {
    pub session_id: String,
    /// 由调用方分配；重复 (session_id, seq) 报错
    pub seq: u64,
    pub role: TurnRole,
    #[serde(default)]
    pub platform_user_id: Option<String>,
    pub content: String,
    #[serde(default)]
    pub attachments: Vec<Attachment>,
    pub created_at: DateTime<Utc>,
}
